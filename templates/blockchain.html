Deployed Smart Contract Architecture (VehicleRegistry.sol)

Solidity ^0.8.0 · EVM Compatible 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract VehicleRegistry {
    struct Vehicle { string vin; string plate; string owner; string status; uint256 timestamp; }
    mapping(string => Vehicle) public vehicles;
    event VehicleRegistered(string indexed plate, string owner, uint256 timestamp);

    function registerVehicle(string memory _vin, string memory _plate, string memory _owner) public {
        vehicles[_plate] = Vehicle(_vin, _plate, _owner, "Active", block.timestamp);
        emit VehicleRegistered(_plate, _owner, block.timestamp);
    }

    function updateStatus(string memory _plate, string memory _newStatus) public {
        require(bytes(vehicles[_plate].plate).length > 0, "Vehicle does not exist");
        vehicles[_plate].status = _newStatus;
    }
}
